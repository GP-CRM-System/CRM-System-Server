# Four Cards(Last six months)

- Total Revenue
- Total Number of Orders
- Conversion Rate
- Cancellation Rate

Date will be handled by the backend(like everything else)

## Aggregate for Total Revenue

```json
db.getCollection('orders').aggregate(
  [
    { $unwind: '$products' },
    {
      $match: {
        createdAt: {
          $gte: ISODate(
            '2025-05-27T12:56:07.000Z'
          )
        }
      }
    },
    {
      $group: {
        _id: null,
        totalRevenue: {
          $sum: {
            $multiply: [
              '$products.unitPrice',
              '$products.quantity'
            ]
          }
        }
      }
    }
  ],
  { maxTimeMS: 60000, allowDiskUse: true }
);
```

db.getCollection('orders').aggregate(

## Aggregate to calculate conversion rate

```json
db.getCollection('contacts').aggregate(
  [
    {
      $match: {
        createdAt: {
          $gte: ISODate(
            '2025-05-27T12:56:07.000Z'
          )
        }
      }
    },
    {
      $facet: {
        twoStages: [
          {
            $match: {
              $expr: {
                $eq: [{ $size: '$stage' }, 2]
              }
            }
          },
          { $count: 'count' }
        ],
        oneStage: [
          {
            $match: {
              $expr: {
                $eq: [{ $size: '$stage' }, 1]
              }
            }
          },
          { $count: 'count' }
        ]
      }
    },
    {
      $project: {
        two: {
          $arrayElemAt: ['$twoStages.count', 0]
        },
        one: {
          $arrayElemAt: ['$oneStage.count', 0]
        }
      }
    },
    {
      $project: {
        percentage: {
          $multiply: [
            { $divide: ['$two', '$one'] },
            100
          ]
        }
      }
    }
  ],
  { maxTimeMS: 60000, allowDiskUse: true }
);
```

## Aggregate to calculate cancellation Rate

```json
db.getCollection('orders').aggregate(
  [
    {
      $match: {
        createdAt: {
          $gte: ISODate(
            '2025-05-27T12:56:07.000Z'
          )
        }
      }
    },
    {
      $facet: {
        cancelledOrders: [
          {
            $match: {
              'stage.stageType': 'Cancelled'
            }
          },
          { $count: 'count' }
        ],
        nonCancelledOrders: [
          {
            $match: {
              'stage.stageType': {
                $ne: 'cancelled'
              }
            }
          },
          { $count: 'count' }
        ]
      }
    },
    {
      $project: {
        percentCancelled: {
          $multiply: [
            {
              $divide: [
                {
                  $arrayElemAt: [
                    '$cancelledOrders.count',
                    0
                  ]
                },
                {
                  $arrayElemAt: [
                    '$nonCancelledOrders.count',
                    0
                  ]
                }
              ]
            },
            100
          ]
        }
      }
    }
  ],
  { maxTimeMS: 60000, allowDiskUse: true }
);
```
