# Lead Conversions

```json
db.getCollection('contacts').aggregate(
  [
    {
      $group: {
        _id: {
          owner: '$owner',
          stagesCount: { $size: '$stage' }
        },
        totalContacts: { $sum: 1 }
      }
    },
    {
      $group: {
        _id: '$_id.owner',
        leadsHandled: { $sum: '$totalContacts' },
        conversions: {
          $sum: {
            $cond: [
              { $eq: ['$_id.stagesCount', 2] },
              '$totalContacts',
              0
            ]
          }
        }
      }
    },
    {
      $project: {
        employeeId: '$_id',
        conversions: 1,
        leadsHandled: 1,
        conversionRate: {
          $multiply: [
            {
              $cond: [
                { $eq: ['$leadsHandled', 0] },
                0,
                {
                  $divide: [
                    '$conversions',
                    '$leadsHandled'
                  ]
                }
              ]
            },
            100
          ]
        }
      }
    }
  ],
  { maxTimeMS: 60000, allowDiskUse: true }
);
```
